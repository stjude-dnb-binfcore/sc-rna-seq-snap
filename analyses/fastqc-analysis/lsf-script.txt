# Thread control (prevent hidden oversubscription)
export OMP_NUM_THREADS=1
export OPENBLAS_NUM_THREADS=1
export MKL_NUM_THREADS=1
export NUMEXPR_NUM_THREADS=1

# JVM thread control
# By default, the JVM decides: number of GC threads ≈ number of detected CPU cores
# So on a node with (for example): 32 cores → JVM might try to use ~32 GC threads
export JAVA_TOOL_OPTIONS="-XX:ParallelGCThreads=1 -XX:ConcGCThreads=1"
